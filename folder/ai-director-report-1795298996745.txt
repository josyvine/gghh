const tools = {
  clamp01(v) {
    return v < 0 ? 0 : v > 1 ? 1 : v;
  },

  lerp(a, b, t) {
    return a + (b - a) * t;
  },

  easeOutCubic(t) {
    const u = 1 - tools.clamp01(t);
    return 1 - u * u * u;
  },

  easeOutExpo(t) {
    const c = tools.clamp01(t);
    return c >= 1 ? 1 : 1 - Math.pow(2, -10 * c);
  },

  easeInOutCubic(t) {
    const c = tools.clamp01(t);
    return c < 0.5 ? 4 * c * c * c : 1 - Math.pow(-2 * c + 2, 3) / 2;
  },

    return 1.0 + bounce + dt * driftRate;

  // GPU WebGL2 stepped top-to-bottom vertical curtain reveal with downward vertical motion smear & warm edge bloom
  renderSteppedVerticalReveal(runtime, sourceCanvas, w, h, revealY, smearStrength = 0.85, bloomStrength = 0.55) {
    const glsl = `
      uniform sampler2D u_tex;
      uniform float u_revealY;
      uniform float u_smear;
      uniform float u_bloom;
      void main() {
        vec2 uv = v_uv;
        // v_uv.y == 1.0 is top in WebGL upright flip, or 0.0 is bottom; map top-down yTop in [0..1]
        float yTop = 1.0 - uv.y;
        float band = 0.06 + 0.16 * u_smear;
        float distPast = yTop - u_revealY;

        if (distPast > band * 1.4) {
          gl_FragColor = vec4(0.0, 0.0, 0.0, 1.0);
          return;
        }

        vec4 baseCol = texture2D(u_tex, uv);

        // Downward vertical motion smear concentrated near and below the leading reveal front
        float frontProximity = smoothstep(-band * 0.85, band * 0.15, distPast);
        float smearAmt = frontProximity * u_smear;

        vec3 accum = vec3(0.0);
        float wSum = 0.0;
        const int SAMPLES = 14;
        for (int i = 0; i < SAMPLES; i++) {
          float fi = float(i) / float(SAMPLES - 1);
          // Pull pixels from above downward to create true downward streak into black
          float sampleYTop = clamp(yTop - fi * band * 1.15 * smearAmt, 0.0, max(0.001, u_revealY));
          vec2 suv = vec2(uv.x, 1.0 - sampleYTop);
          float wt = 1.0 - fi * 0.65;
          accum += texture2D(u_tex, suv).rgb * wt;
          wSum += wt;
        }
        vec3 smeared = accum / max(0.0001, wSum);
        vec3 col = mix(baseCol.rgb, smeared, clamp(smearAmt * 1.15, 0.0, 1.0));

        // Warm highlight bloom along active vertical smear front
        float luma = dot(col, vec3(0.299, 0.587, 0.114));
        float glowBand = exp(-pow(distPast / max(0.02, band * 0.55), 2.0));
        vec3 warmGlow = col * smoothstep(0.32, 0.92, luma) * glowBand * u_bloom * (0.4 + 0.9 * u_smear) * vec3(1.18, 1.04, 0.86);
        col += warmGlow;

        // Softly attenuate into pure black past revealY
        float mask = 1.0 - smoothstep(0.0, max(0.008, band * (0.35 + 0.85 * u_smear)), distPast);
        col *= mask;

        gl_FragColor = vec4(col, 1.0);
      }
    `;
    return runtime.runShader(
      'steppedVerticalReveal',
      w,
      h,
      glsl,
      { u_tex: sourceCanvas },
      {
        u_revealY: revealY,
        u_smear: tools.clamp01(smearStrength),
        u_bloom: bloomStrength
      }
    );
  },

        gl_FragColor = vec4(base, 1.0);
      }

  // Additive 10-strand swirling tapered glowing light trails + floating orb particles (11.90s..13.00s & 15.85s..16.08s)
  // Strictly proportional to (w, h), NEVER wipes main canvas
  drawSwirlingLightTrails(ctx, w, h, localT, duration = 1.0, intensity = 1.0, burstMode = false) {
    if (intensity <= 0.001) return;
    const minDim = Math.min(w, h);
    const p = tools.clamp01(localT / Math.max(0.05, duration));
    const envelope = burstMode
      ? Math.sin(p * Math.PI) * 1.15
      : Math.sin(Math.min(1, p * 1.15) * Math.PI) * 0.95;
    const alphaScale = tools.clamp01(envelope * intensity);
    if (alphaScale <= 0.002) return;

    const strandCount = burstMode ? 11 : 9;
    const steps = 36;

    ctx.save();
    ctx.globalCompositeOperation = 'lighter';
    ctx.lineCap = 'round';
    ctx.lineJoin = 'round';

    for (let s = 0; s < strandCount; s++) {
      const seed = s * 1.73205 + (burstMode ? 11.3 : 2.7);
      const phase = p * (burstMode ? 1.8 : 1.35) + (s % 4) * 0.14;
      const head = tools.clamp01(phase * 1.25 - 0.12);
      const tailLen = burstMode ? 0.68 : 0.56;
      const tail = Math.max(0.0, head - tailLen);
      if (head - tail < 0.03) continue;

      // Anchor curves organically across frame (edges + diagonal sweeps)
      const leftToRight = (s % 2) === 0;
      const x0 = leftToRight ? -0.08 * w : 1.08 * w;
      const x3 = leftToRight ? 1.08 * w : -0.08 * w;
      const yBase = ((s + 0.5) / strandCount) * h;
      const ampY = h * (burstMode ? 0.34 : 0.26);
      const y0 = yBase + Math.sin(seed * 1.9) * ampY * 0.6;
      const y3 = h - yBase + Math.cos(seed * 2.3) * ampY * 0.55;

      const cx1 = w * (0.25 + 0.35 * Math.sin(seed + localT * 1.4));
      const cy1 = y0 + Math.cos(seed * 1.3 + localT * 1.9) * ampY;
      const cx2 = w * (0.75 + 0.35 * Math.cos(seed * 0.8 - localT * 1.5));
      const cy2 = y3 + Math.sin(seed * 1.7 - localT * 1.6) * ampY;

      const pts = [];
      for (let i = 0; i <= steps; i++) {
        const u = tail + (head - tail) * (i / steps);
        const inv = 1 - u;
        const bx = inv * inv * inv * x0 + 3 * inv * inv * u * cx1 + 3 * inv * u * u * cx2 + u * u * u * x3;
        const by = inv * inv * inv * y0 + 3 * inv * inv * u * cy1 + 3 * inv * u * u * cy2 + u * u * u * y3;
        const wave = Math.sin(u * 9.0 + seed + localT * 4.2) * minDim * 0.022;
        pts.push({ x: bx + wave * 0.45, y: by + wave, frac: i / steps });
      }

      // Draw tapered multi-pass additive ribbon (outer soft glow + crisp bright core)
      const baseWidth = minDim * (burstMode ? 0.0055 : 0.0042) * (0.7 + 0.55 * ((s * 37) % 10) / 10);
      for (let i = 1; i < pts.length; i++) {
        const a = pts[i - 1];
        const b = pts[i];
        const taper = Math.sin(b.frac * Math.PI);
        if (taper <= 0.01) continue;

        // Outer soft warm-white bloom
        ctx.strokeStyle = `rgba(255, 245, 232, ${(0.22 * taper * alphaScale).toFixed(3)})`;
        ctx.lineWidth = Math.max(1, baseWidth * 3.8 * taper);
        ctx.beginPath();
        ctx.moveTo(a.x, a.y);
        ctx.lineTo(b.x, b.y);
        ctx.stroke();

        // Crisp white core
        ctx.strokeStyle = `rgba(255, 255, 255, ${(0.82 * taper * alphaScale).toFixed(3)})`;
        ctx.lineWidth = Math.max(0.75, baseWidth * 1.15 * taper);
        ctx.beginPath();
        ctx.moveTo(a.x, a.y);
        ctx.lineTo(b.x, b.y);
        ctx.stroke();
      }

      // Bright glowing orb/sparkle along each strand
      const orbIdx = Math.floor(pts.length * (0.55 + 0.3 * Math.sin(seed + localT * 2.5)));
      const orb = pts[Math.max(0, Math.min(pts.length - 1, orbIdx))];
      const orbR = minDim * (burstMode ? 0.014 : 0.010);
      const grad = ctx.createRadialGradient(orb.x, orb.y, 0, orb.x, orb.y, orbR * 2.2);
      grad.addColorStop(0, `rgba(255, 255, 255, ${(0.95 * alphaScale).toFixed(3)})`);
      grad.addColorStop(0.35, `rgba(255, 246, 230, ${(0.45 * alphaScale).toFixed(3)})`);
      grad.addColorStop(1, 'rgba(255, 240, 220, 0)');
      ctx.fillStyle = grad;
      ctx.beginPath();
      ctx.arc(orb.x, orb.y, orbR * 2.2, 0, Math.PI * 2);
      ctx.fill();
    }

    ctx.restore();
  },

  // GPU WebGL2 warm cinematic contrast color grade + soft vignette (applied cleanly as final pass)
  renderWarmGrade(runtime, sourceCanvas, w, h, opts = {}) {
    const warmth = opts.warmth !== undefined ? opts.warmth : 0.12;
    const contrast = opts.contrast !== undefined ? opts.contrast : 1.08;
    const saturation = opts.saturation !== undefined ? opts.saturation : 1.10;
    const vignette = opts.vignette !== undefined ? opts.vignette : 0.24;
    const flash = opts.flash !== undefined ? opts.flash : 0.0;

    const glsl = `
      uniform sampler2D u_tex;
      uniform float u_warmth;
      uniform float u_contrast;
      uniform float u_saturation;
      uniform float u_vignette;
      uniform float u_flash;
      void main() {
        vec2 uv = v_uv;
        vec3 col = texture2D(u_tex, uv).rgb;

        // Contrast curve around mid-tones
        col = (col - 0.5) * u_contrast + 0.5;

        // Rich warm amber tint
        col *= vec3(1.0 + u_warmth * 0.55, 1.0 + u_warmth * 0.12, 1.0 - u_warmth * 0.55);

        // Saturation lift
        float luma = dot(col, vec3(0.299, 0.587, 0.114));
        col = mix(vec3(luma), col, u_saturation);

        // Subtle beat flash boost
        col += vec3(u_flash);

        // Soft corner vignette
        vec2 d = (uv - 0.5) * vec2(1.15, 1.0);
        float vig = 1.0 - smoothstep(0.45, 0.98, length(d)) * u_vignette;
        col *= vig;

        gl_FragColor = vec4(clamp(col, 0.0, 1.0), 1.0);
      }
    `;
    return runtime.runShader(
      'warmGrade',
      w,
      h,
      glsl,
      { u_tex: sourceCanvas },
      {
        u_warmth: warmth,
        u_contrast: contrast,
        u_saturation: saturation,
        u_vignette: vignette,
        u_flash: flash
      }
    );
  },

  // GPU WebGL2 outro heavy defocus + warm-gray wash fading cleanly to pure black (20.40s..21.85s)
  renderOutroDefocusWash(runtime, sourceCanvas, w, h, progress) {
    const p = tools.clamp01(progress);
    const glsl = `
      uniform sampler2D u_tex;
      uniform float u_prog;
      void main() {
        vec2 uv = v_uv;
        float blurRad = smoothstep(0.0, 0.35, u_prog) * 0.085;
        vec3 acc = vec3(0.0);
        float total = 0.0;
        const int RINGS = 3;
        const int DIRS = 8;
        acc += texture2D(u_tex, uv).rgb;
        total += 1.0;
        for (int r = 1; r <= RINGS; r++) {
          float rad = blurRad * (float(r) / float(RINGS));
          for (int d = 0; d < DIRS; d++) {
            float ang = float(d) * 0.78539816;
            vec2 off = vec2(cos(ang), sin(ang)) * rad;
            acc += texture2D(u_tex, clamp(uv + off, vec2(0.002), vec2(0.998))).rgb;
            total += 1.0;
          }
        }
        vec3 blurred = acc / total;

        // Warm-neutral gray gradient wash (peaks around p = 0.25..0.45 like 20.60s)
        float washAmt = smoothstep(0.05, 0.32, u_prog);
        vec3 grayGrad = mix(vec3(0.54, 0.52, 0.48), vec3(0.26, 0.26, 0.24), uv.y);
        vec3 col = mix(blurred, grayGrad, washAmt * 0.88);

        // Smooth fade to pure black from p = 0.35..0.92
        float fadeBlack = 1.0 - smoothstep(0.32, 0.90, u_prog);
        col *= fadeBlack;

        gl_FragColor = vec4(col, 1.0);
      }
    `;
    return runtime.runShader(
      'outroDefocusWash',
      w,
      h,
      glsl,
      { u_tex: sourceCanvas },
      { u_prog: p }
    );
  }
};

function clamp01(v) {
  return v < 0 ? 0 : (v > 1 ? 1 : v);
}

function lerp(a, b, p) {
  return a + (b - a) * p;
}

function easeOutCubic(x) {
  const t = clamp01(x);
  return 1 - Math.pow(1 - t, 3);
}

function easeInOutCubic(x) {
  const t = clamp01(x);
  return t < 0.5 ? 4 * t * t * t : 1 - Math.pow(-2 * t + 2, 3) / 2;
}

function beatZoomScale(localT, peakZoom, decaySec, driftRate) {
  const dt = Math.max(0, localT);
  const decay = Math.exp(-dt / Math.max(0.015, (decaySec || 0.24) * 0.38));
  return 1.0 + (peakZoom - 1.0) * decay + dt * (driftRate || 0);
}

    g.addColorStop(0, '#000000');
    g.addColorStop(1, '#ffffff');
    c.fillStyle = g;
    c.fillRect(0, 0, 4, 64);
    revealGradReady = true;
  }
  return canvas;
}

// Beat-synced stepped top-to-bottom reveal aligned strictly to detected beats: 0.40, 0.90, 1.40, 1.95, 2.70, 3.42
function getSteppedRevealState(t) {
  const steps = [
    { start: 0.36, dur: 0.22, fromY: 0.00, toY: 0.14 },
    { start: 0.86, dur: 0.22, fromY: 0.14, toY: 0.24 },
    { start: 1.36, dur: 0.24, fromY: 0.24, toY: 0.37 },
    { start: 1.91, dur: 0.28, fromY: 0.37, toY: 0.54 },
    { start: 2.66, dur: 0.30, fromY: 0.54, toY: 0.75 },
    { start: 3.36, dur: 0.24, fromY: 0.75, toY: 1.04 }
  ];

  if (t < steps[0].start) {
    return { revealY: 0.0, edgePulse: 0.0 };
  }

  let revealY = 0.0;
  let edgePulse = 0.18;

  for (let i = 0; i < steps.length; i++) {
    const s = steps[i];
    if (t < s.start) {
      revealY = s.fromY;
      // Gentle residual soft smear even during short holds (matches REF 0.9s / 1.95s / 2.7s)
      const prevEnd = i > 0 ? steps[i - 1].start + steps[i - 1].dur : 0;
      const holdDecay = Math.exp(-Math.max(0, t - prevEnd) / 0.32);
      edgePulse = 0.16 + 0.34 * holdDecay;
      break;
    } else if (t <= s.start + s.dur) {
      const p = clamp01((t - s.start) / s.dur);
      revealY = lerp(s.fromY, s.toY, easeOutCubic(p));
      edgePulse = 0.35 + 0.65 * Math.sin(p * Math.PI);
      break;
    } else if (i === steps.length - 1) {
      revealY = s.toY;
      edgePulse = 0.0;
    }
  }
  return { revealY, edgePulse };
}

// 2D anisotropic downward motion-smear shader (eliminates vertical barcode banding & harsh horizontal bar)
const SHADER_STEPPED_REVEAL = `
precision highp float;
varying vec2 v_uv;
uniform sampler2D u_src;
uniform sampler2D u_grad;
uniform float u_revealY;
uniform float u_smear;
uniform float u_bloom;

void main() {
  float yNorm = texture2D(u_grad, v_uv).r;
  float yDir = texture2D(u_grad, vec2(0.5, 0.6)).r > texture2D(u_grad, vec2(0.5, 0.4)).r ? 1.0 : -1.0;

  if (u_revealY <= 0.002) {
    gl_FragColor = vec4(0.0, 0.0, 0.0, 1.0);
    return;
  }

  float smearBand = 0.045 + 0.19 * u_smear;
  float edgeDist = yNorm - u_revealY;

  if (edgeDist > smearBand * 1.65) {
    gl_FragColor = vec4(0.0, 0.0, 0.0, 1.0);
    return;
  }

  vec3 crisp = texture2D(u_src, v_uv).rgb;

  // Soft 2D diffused downward smear near and below revealY (no sharp vertical stripes)
  float blurZone = smoothstep(-smearBand * 0.75, smearBand * 0.25, edgeDist);
  vec3 smearCol = vec3(0.0);
  float wSum = 0.0;

  // 3 horizontal taps x 7 vertical taps pulled smoothly from above revealY
  for (int ix = -1; ix <= 1; ix++) {
    float hx = float(ix) * (0.012 + 0.016 * u_smear) * blurZone;
    float wx = (ix == 0) ? 1.0 : 0.68;
    for (int iy = 0; iy < 7; iy++) {
      float fi = float(iy) / 6.0;
      float baseY = min(yNorm, u_revealY);
      float sampleY = clamp(baseY - fi * (0.045 + 0.085 * u_smear), 0.002, 0.998);
      float sv = yDir > 0.0 ? sampleY : (1.0 - sampleY);
      vec2 suv = vec2(clamp(v_uv.x + hx, 0.002, 0.998), clamp(sv, 0.002, 0.998));
      float wy = 1.0 - fi * 0.45;
      smearCol += texture2D(u_src, suv).rgb * (wx * wy);
      wSum += wx * wy;
    }
  }
  smearCol /= max(wSum, 0.001);

  vec3 col = mix(crisp, smearCol, clamp(blurZone * (0.35 + 0.65 * u_smear), 0.0, 1.0));

  // Highlight-driven warm bloom along the soft smear front (only blooms bright image areas, never a flat bar)
  float luma = dot(smearCol, vec3(0.299, 0.587, 0.114));
  float frontEnv = exp(-pow((edgeDist + smearBand * 0.1) / max(0.025, smearBand * 0.65), 2.0));
  float activeMask = smoothstep(0.01, 0.05, u_revealY) * (1.0 - smoothstep(0.95, 1.03, u_revealY));
  vec3 highlightBloom = smearCol * smoothstep(0.28, 0.85, luma) * frontEnv * u_bloom * 0.85 * vec3(1.18, 1.02, 0.82) * activeMask;
  col += highlightBloom;

  // Smooth exponential fade into pure black below revealY
  float mask = 1.0 - smoothstep(-0.01, max(0.015, smearBand * 1.25), edgeDist);
  // Extra soft tail for dreamy vertical falloff
  float tailMask = edgeDist > 0.0 ? exp(-edgeDist / max(0.012, smearBand * 0.48)) : 1.0;
  col *= mix(mask, tailMask, 0.55);

  gl_FragColor = vec4(clamp(col, 0.0, 1.0), 1.0);
}
`;

const SHADER_DIR_BLUR = `
precision highp float;
varying vec2 v_uv;
uniform sampler2D u_src;
uniform float u_dirX;
uniform float u_dirY;
uniform float u_rad;

void main() {
  vec2 dir = vec2(u_dirX, u_dirY);
  vec2 centerVec = (v_uv - vec2(0.5, 0.5)) * u_rad;
  vec3 acc = vec3(0.0);
  float wSum = 0.0;
  for (int i = -7; i <= 7; i++) {
    float fi = float(i) / 7.0;
    float w = 1.0 - abs(fi) * 0.58;
    vec2 suv = clamp(v_uv + dir * fi + centerVec * fi, vec2(0.002), vec2(0.998));
    acc += texture2D(u_src, suv).rgb * w;
    wSum += w;
  }
  gl_FragColor = vec4(acc / max(wSum, 0.001), 1.0);
}
`;

const SHADER_SPLIT_WIPE = `
precision highp float;
varying vec2 v_uv;
uniform sampler2D u_left;
uniform sampler2D u_right;
uniform float u_splitX;
uniform float u_seamW;
uniform float u_seamAmt;

void main() {
  float dist = abs(v_uv.x - u_splitX);
  float blurFactor = u_seamW > 0.001 ? exp(-(dist * dist) / max(0.0001, 2.0 * u_seamW * u_seamW)) * u_seamAmt : 0.0;
  float edgeSoft = max(0.0015, u_seamW * 0.22);
  float maskR = smoothstep(u_splitX - edgeSoft, u_splitX + edgeSoft, v_uv.x);

  vec3 col = vec3(0.0);
  if (blurFactor > 0.01) {
    vec3 acc = vec3(0.0);
    float wSum = 0.0;
    float span = 0.085 * blurFactor;
    for (int i = -6; i <= 6; i++) {
      float fi = float(i) / 6.0;
      float w = 1.0 - abs(fi) * 0.5;
      vec2 suv = vec2(clamp(v_uv.x + fi * span, 0.001, 0.999), v_uv.y);
      float m = smoothstep(u_splitX - edgeSoft, u_splitX + edgeSoft, suv.x);
      vec3 cL = texture2D(u_left, suv).rgb;
      vec3 cR = texture2D(u_right, suv).rgb;
      acc += mix(cL, cR, m) * w;
      wSum += w;
    }
    col = acc / max(wSum, 0.001);
  } else {
    vec3 cL = texture2D(u_left, v_uv).rgb;
    vec3 cR = texture2D(u_right, v_uv).rgb;
    col = mix(cL, cR, maskR);
  }

  gl_FragColor = vec4(col, 1.0);
}
`;

// Rich cohesive warm golden-amber cinematic color grade with filmic S-curve & split toning
const SHADER_WARM_GRADE = `
precision highp float;
varying vec2 v_uv;
uniform sampler2D u_src;
uniform float u_warmth;
uniform float u_contrast;
uniform float u_saturation;
uniform float u_vignette;
uniform float u_exposure;
uniform float u_tintGreen;
uniform float u_flash;

void main() {
  vec3 col = texture2D(u_src, v_uv).rgb;

  // Exposure normalization
  col *= u_exposure;

  // Subtle green-magenta neutralization + golden-amber warm balance
  col.r *= (1.0 + u_warmth * 0.48);
  col.g *= (1.0 + u_warmth * 0.14 + u_tintGreen);
  col.b *= (1.0 - u_warmth * 0.44);

  // Filmic S-curve contrast with rich deep shadows
  vec3 sc = col * col * (3.0 - 2.0 * col);
  col = mix(col, sc, 0.24);
  col = (col - vec3(0.48)) * u_contrast + vec3(0.48);
  col = clamp(col, 0.0, 1.0);

  // Warm golden highlight & espresso shadow split-toning
  float luma = dot(col, vec3(0.2126, 0.7152, 0.0722));
  vec3 shadowTone = vec3(0.11, 0.07, 0.04);
  vec3 highTone = vec3(1.06, 0.98, 0.86);
  col = mix(col, col * highTone, smoothstep(0.35, 0.95, luma) * 0.42);
  col += shadowTone * (1.0 - smoothstep(0.0, 0.40, luma)) * 0.18;

  // Controlled saturation
  float luma2 = dot(col, vec3(0.2126, 0.7152, 0.0722));
  col = mix(vec3(luma2), col, u_saturation);

  // Soft atmospheric corner vignette matching reference
  vec2 d = (v_uv - vec2(0.5)) * vec2(1.18, 1.02);
  float vig = 1.0 - smoothstep(0.28, 0.96, length(d)) * u_vignette;
  col *= clamp(vig, 0.42, 1.0);

  // Subtle beat flash lift
  col += vec3(1.0, 0.95, 0.88) * u_flash;

  gl_FragColor = vec4(clamp(col, 0.0, 1.0), 1.0);
}
`;

// Outro heavy isotropic defocus + warm-gray gradient wash matching REF 20.60s & pure black by REF 21.50s
const SHADER_OUTRO_DEFOCUS = `
precision highp float;
varying vec2 v_uv;
uniform sampler2D u_src;
uniform sampler2D u_grad;
uniform float u_prog;
uniform float u_aspect;

void main() {
  float yNorm = texture2D(u_grad, v_uv).r; // 0 = top, 1 = bottom
  float blurP = smoothstep(0.0, 0.24, u_prog);
  float radius = 0.015 + 0.115 * blurP;
  vec2 invAspect = vec2(1.0 / max(0.25, u_aspect), 1.0);

  vec3 acc = texture2D(u_src, v_uv).rgb * 0.10;
  float wSum = 0.10;

  for (int r = 1; r <= 3; r++) {
    float rr = radius * (float(r) / 3.0);
    float rw = 1.0 - float(r) * 0.22;
    for (int i = 0; i < 8; i++) {
      float ang = float(i) * 0.78539816 + float(r) * 0.26;
      vec2 off = vec2(cos(ang), sin(ang)) * rr * invAspect;
      acc += texture2D(u_src, clamp(v_uv + off, vec2(0.002), vec2(0.998))).rgb * rw;
      wSum += rw;
    }
  }

  vec3 blurred = acc / max(wSum, 0.001);

  // Warm-neutral vertical gray wash: darker warm charcoal at top, lighter warm taupe at bottom (REF 20.60s 1:1)
  vec3 grayTop = vec3(0.24, 0.23, 0.20);
  vec3 grayBot = vec3(0.56, 0.53, 0.48);
  vec3 warmGrayGrad = mix(grayTop, grayBot, smoothstep(0.0, 1.0, yNorm));

  // Fully reaches smooth defocused gray wash by 20.55s-20.60s (prog ~0.20)
  float washAmt = smoothstep(0.03, 0.22, u_prog);
  vec3 col = mix(blurred, warmGrayGrad, washAmt * 0.94);

  // Smooth fade from warm gray wash to pure black matching REF 21.40s-21.50s (prog ~0.78)
  float fadeOut = 1.0 - smoothstep(0.26, 0.80, u_prog);
  col *= fadeOut;

  gl_FragColor = vec4(clamp(col, 0.0, 1.0), 1.0);
}
`;

function drawBorderlessQuadCollage(ctx, clips, panels, width, height) {
  for (let i = 0; i < panels.length; i++) {
    const p = panels[i];
    if (!p || p.alpha <= 0.001) continue;
    const rx = p.rect[0] * width;
    const ry = p.rect[1] * height;
    const rw = p.rect[2] * width;
    const rh = p.rect[3] * height;

    const media = p.overrideMedia || getCachedMedia(clips, p.clipIdx);
    if (!media) continue;
    const meta = p.overrideMedia ? { anchorX: 0.5, anchorY: 0.5 } : getClipMeta(p.clipIdx);

    ctx.save();
    ctx.globalAlpha = clamp01(p.alpha);
    drawCoverAnchored(ctx, media, rx, ry, rw, rh, p.zoom || 1.0, meta.anchorX, meta.anchorY);
    ctx.restore();
  }
}

// Immediately copies each segmentSubject() result into its own isolated offscreen canvas to prevent shared-buffer aliasing
async function extractIsolatedCutout(srcBuf, copyKey, width, height, feather = 0.20) {
  const seg = await runtime.segmentSubject(srcBuf, width, height, feather);
  const copyCanvas = runtime.getCanvas(copyKey, width, height);
  const g = copyCanvas.getContext('2d');
  g.clearRect(0, 0, width, height);
  if (seg) {
    g.drawImage(seg, 0, 0, width, height);
  }
  return copyCanvas;
}

async function ensureCutouts(clips, width, height) {
  const key = `${width}x${height}`;
  if (cutoutCacheKey === key && cutoutHeroA && cutoutHeroF && cutoutSubD) return;
  cutoutCacheKey = key;

  // Hero A (#2): matches exact scale & anchorY=0.22 of fullscreen Hero A
  const bufA = prepareCoverBuffer('segSrcA', clips, 2, width, height, 1.04);
  cutoutHeroA = await extractIsolatedCutout(bufA, 'cutoutCopy_A', width, height, 0.22);

  // Hero F (#12): matches exact scale & anchorY=0.14 of fullscreen Hero F
  const bufF = prepareCoverBuffer('segSrcF', clips, 12, width, height, 1.02);
  cutoutHeroF = await extractIsolatedCutout(bufF, 'cutoutCopy_F', width, height, 0.22);

  // Seated D (#5): matches exact scale & anchorY=0.46 of Seated Portrait D
  const bufD = prepareCoverBuffer('segSrcD', clips, 5, width, height, 1.0);
  cutoutSubD = await extractIsolatedCutout(bufD, 'cutoutCopy_D', width, height, 0.18);
}

function drawCrispRimCutout(ctx, cutoutCanvas, width, height, opts = {}) {
  if (!cutoutCanvas) return;
  const scale = opts.scale !== undefined ? opts.scale : 1.0;
  const offsetX = (opts.offsetXNorm || 0) * width;
  const offsetY = (opts.offsetYNorm || 0) * height;
  const alpha = opts.alpha !== undefined ? clamp01(opts.alpha) : 1.0;
  const rimStrength = opts.rimStrength !== undefined ? opts.rimStrength : 0;
  if (alpha <= 0.001) return;

  const dw = width * scale;
  const dh = height * scale;
  const dx = (width - dw) * 0.5 + offsetX;
  const dy = (height - dh) * 0.5 + offsetY;

  ctx.save();
  ctx.globalAlpha = alpha;
  if (rimStrength > 0.001) {
    const minDim = Math.min(width, height);
    ctx.shadowColor = `rgba(255, 245, 225, ${(0.88 * Math.min(1, rimStrength)).toFixed(3)})`;
    ctx.shadowBlur = Math.max(2, minDim * 0.016 * rimStrength);
    ctx.drawImage(cutoutCanvas, dx, dy, dw, dh);
    ctx.shadowBlur = 0;
  }
  ctx.drawImage(cutoutCanvas, dx, dy, dw, dh);
  ctx.restore();
}

function drawRevealLowerLeftGlow(ctx, width, height, intensity) {
  if (intensity <= 0.005) return;
  ctx.save();
  ctx.globalCompositeOperation = 'lighter';
  const gx = width * 0.22;
  const gy = height * 0.89;
  const rx = width * 0.08;
  const ry = height * 0.24;
  ctx.translate(gx, gy);
  ctx.scale(1, ry / Math.max(1, rx));
  const grad = ctx.createRadialGradient(0, 0, 0, 0, 0, rx);
  grad.addColorStop(0, `rgba(255, 250, 236, ${(0.82 * intensity).toFixed(3)})`);
  grad.addColorStop(0.45, `rgba(255, 228, 188, ${(0.36 * intensity).toFixed(3)})`);
  grad.addColorStop(1, 'rgba(255, 205, 150, 0)');
  ctx.fillStyle = grad;
  ctx.beginPath();
  ctx.arc(0, 0, rx, 0, Math.PI * 2);
  ctx.fill();
  ctx.restore();
}

    { p0: [0.06, 1.06],  c1: [0.42, 0.62], c2: [0.76, 0.84], p3: [1.06, 0.56], phase: 0.05, w: 0.85, wave: 0.019 },
    { p0: [0.28, -0.05], c1: [0.54, 0.24], c2: [0.24, 0.58], p3: [0.76, 1.06], phase: 0.03, w: 0.8, wave: 0.022 },
    { p0: [1.06, 0.86],  c1: [0.66, 0.94], c2: [0.32, 0.54], p3: [-0.06, 0.82], phase: 0.04, w: 0.9, wave: 0.020 }
  ];

  const strands = burstMode ? burstStrands : ambientStrands;
  const steps = 38;

  for (let s = 0; s < strands.length; s++) {
    const st = strands[s];
    const speed = burstMode ? 1.55 : 1.32;
    const head = clamp01(prog * speed + (burstMode ? 0.22 : 0.08) - st.phase);
    const tailLen = burstMode ? 0.82 : 0.66;
    const tail = clamp01(head - tailLen);
    if (head - tail < 0.04) continue;

    const drift = localT * (burstMode ? 3.2 : 1.8) + s * 1.37;
    const x0 = st.p0[0] * width;
    const y0 = st.p0[1] * height;
    const cx1 = (st.c1[0] + Math.sin(drift) * 0.035) * width;
    const cy1 = (st.c1[1] + Math.cos(drift * 1.1) * 0.035) * height;
    const cx2 = (st.c2[0] + Math.cos(drift * 0.9) * 0.035) * width;
    const cy2 = (st.c2[1] + Math.sin(drift * 1.2) * 0.035) * height;
    const x3 = st.p3[0] * width;
    const y3 = st.p3[1] * height;

    const pts = [];
    for (let i = 0; i <= steps; i++) {
      const frac = i / steps;
      const u = lerp(tail, head, frac);
      const inv = 1 - u;
      const bx = inv * inv * inv * x0 + 3 * inv * inv * u * cx1 + 3 * inv * u * u * cx2 + u * u * u * x3;
      const by = inv * inv * inv * y0 + 3 * inv * inv * u * cy1 + 3 * inv * u * u * cy2 + u * u * u * y3;
      const curl = Math.sin(u * 8.5 + drift) * minDim * st.wave;
      const curl2 = Math.cos(u * 13.0 - drift * 0.8) * minDim * st.wave * 0.45;
      pts.push({ x: bx + curl * 0.55 + curl2 * 0.3, y: by + curl + curl2 * 0.5, frac });
    }

    const baseCore = minDim * (burstMode ? 0.0024 : 0.0019) * st.w;

    // Tapered multi-pass wispy strokes
    for (let i = 1; i < pts.length; i++) {
      const a = pts[i - 1];
      const b = pts[i];
      const taper = Math.pow(Math.sin(b.frac * Math.PI), 0.72);
      if (taper <= 0.015) continue;

      // Soft wide warm-white atmospheric halo
      ctx.strokeStyle = `rgba(255, 242, 224, ${(0.15 * taper * strength).toFixed(3)})`;
      ctx.lineWidth = Math.max(1.0, baseCore * 5.2 * taper);
      ctx.beginPath();
      ctx.moveTo(a.x, a.y);
      ctx.lineTo(b.x, b.y);
      ctx.stroke();

      // Delicate mid glow
      ctx.strokeStyle = `rgba(255, 250, 244, ${(0.38 * taper * strength).toFixed(3)})`;
      ctx.lineWidth = Math.max(0.8, baseCore * 2.2 * taper);
      ctx.beginPath();
      ctx.moveTo(a.x, a.y);
      ctx.lineTo(b.x, b.y);
      ctx.stroke();

      // Razor-fine luminous white filament core
      ctx.strokeStyle = `rgba(255, 255, 255, ${(0.88 * taper * strength).toFixed(3)})`;
      ctx.lineWidth = Math.max(0.6, baseCore * 0.95 * taper);
      ctx.beginPath();
      ctx.moveTo(a.x, a.y);
      ctx.lineTo(b.x, b.y);
      ctx.stroke();
    }

    // Soft glowing stardust orbs floating along each strand
    const orbFracs = burstMode ? [0.35, 0.72] : [0.58];
    for (let k = 0; k < orbFracs.length; k++) {
      const idx = Math.floor(clamp01(orbFracs[k] + 0.14 * Math.sin(drift + k * 2.1)) * (pts.length - 1));
      const orb = pts[idx];
      const orbR = minDim * (burstMode ? 0.012 : 0.0085) * (0.75 + 0.45 * ((s + k) % 3) * 0.35);
      const grad = ctx.createRadialGradient(orb.x, orb.y, 0, orb.x, orb.y, orbR * 2.2);
      grad.addColorStop(0, `rgba(255, 255, 255, ${(0.95 * strength).toFixed(3)})`);
      grad.addColorStop(0.32, `rgba(255, 246, 232, ${(0.42 * strength).toFixed(3)})`);
      grad.addColorStop(1, 'rgba(255, 238, 215, 0)');
      ctx.fillStyle = grad;
      ctx.beginPath();
      ctx.arc(orb.x, orb.y, orbR * 2.2, 0, Math.PI * 2);
      ctx.fill();
    }
  }

  ctx.restore();
}

async function init(clips) {
  cachedFrames = [];
  cutoutCacheKey = '';
  cutoutHeroA = null;
  cutoutHeroF = null;
  cutoutSubD = null;
  revealGradReady = false;

  if (!clips || clips.length === 0) return;
  for (let i = 0; i < clips.length; i++) {

    cachedFrames.push(frame || clips[i]);
  }
}

    } else if (seg.blurType === 'vertShake') {
      const distShake = Math.abs(t - seg.shakeAt);
      const shakeEnv = Math.exp(-(distShake * distShake) / (2 * 0.085 * 0.085));
      blurAngle = Math.PI * 0.5;
      dirBlur = seg.blurAmt * shakeEnv;
      radBlur = 0.028 * shakeEnv;
      zoom += 0.055 * shakeEnv;
      offsetY = Math.sin(localT * 42.0) * 0.01 * shakeEnv;
    }

    const srcM = prepareCoverBuffer('montageSrc', clips, seg.clipIdx, width, height, zoom, 0, offsetY);
    if (dirBlur > 0.003 || radBlur > 0.003) {
      const blurred = renderDirectionalBlur(srcM, width, height, blurAngle, dirBlur, radBlur);
      compCtx.drawImage(blurred, 0, 0, width, height);
    } else {
      compCtx.drawImage(srcM, 0, 0, width, height);
    }

    const meta = getClipMeta(seg.clipIdx);
    gradeOpts = getGradeForProfile(meta.profile, { flash });

    if (seg.lightBurst) {
      postLightTrails = {
        localT: localT + 0.04,
        duration: 0.28,
        intensity: 1.25,
        burstMode: true
      };
    }
  } else {
    // Segment 24 (20.25s..end): Rapid heavy defocus into warm-gray wash by 20.60s -> pure black by 21.50s
    applyGrade = false;
    const prog = clamp01((t - 20.25) / Math.max(0.05, duration - 20.25));
    const zoom = 1.03 + prog * 0.06;
    const srcOut = prepareCoverBuffer('outroSrc', clips, 2, width, height, zoom);
    const gradedOut = renderWarmGrade(srcOut, width, height, getGradeForProfile('warm'));
    const defocused = renderOutroDefocusWash(gradedOut, width, height, prog);
    compCtx.drawImage(defocused, 0, 0, width, height);
  }

  // Final cohesive warm cinematic color grade pass onto main ctx
  if (applyGrade) {
    const graded = renderWarmGrade(compCanvas, width, height, gradeOpts);
    ctx.drawImage(graded, 0, 0, width, height);
  } else {
    ctx.drawImage(compCanvas, 0, 0, width, height);
  }

  // Draw delicate wispy additive light trails post-grade so white luminous strands stay crisp
  if (postLightTrails) {
    drawWispyLightTrails(
      ctx,
      width,
      height,
      postLightTrails.localT,
      postLightTrails.duration,
      postLightTrails.intensity,
      postLightTrails.burstMode
    );
  }
}
